# The C ABI the apps link. Everything below it is C++; this is the only module
# that speaks C, and the only one whose declarations are not ours to change
# freely: include/analyzer.h is the contract with the macOS and iOS apps.
#
# That header is plain C, written for Swift's importer, and a C++ translation
# unit cannot include it as it stands: with C++ guards off, an enum is declared
# as `enum X { ... }` followed by `typedef uint32_t X`, which C++ rejects as a
# conflict. And a definition marked `noexcept` - which every entry point must
# be, since nothing may unwind into C - cannot follow a declaration that is not.
#
# So the implementation compiles against a C++ view of the same header,
# produced here from include/analyzer.h by two mechanical rewrites and never
# edited by hand:
#
#   - the C23 branch of each enum (`enum X : uint32_t`) is taken, which is
#     valid C++, instead of the typedef branch;
#   - every function declaration gains `noexcept`.
#
# Because the definitions in src/ffi/*.cpp see the real declarations, a
# signature that drifts from the header is a compile error, not a link-time
# surprise in somebody else's Swift.

set(analyzer_c_header ${PROJECT_SOURCE_DIR}/include/analyzer.h)
set(analyzer_ffi_generated ${CMAKE_CURRENT_BINARY_DIR}/generated)

# Regenerate when the header changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${analyzer_c_header})

file(READ ${analyzer_c_header} analyzer_header_text)
string(REPLACE "__STDC_VERSION__ >= 202311L" "1" analyzer_header_text "${analyzer_header_text}")
string(REPLACE ");" ") noexcept;" analyzer_header_text "${analyzer_header_text}")

# Every function declaration starts a line, has an `analyzer_` name and is
# followed by `(`; every one of them should have been rewritten. A mismatch
# means the header grew a shape the rewrite does not understand.
string(REGEX MATCHALL "\n[A-Za-z_][^\n(]*analyzer_[a-z0-9_]+\\(" analyzer_declarations
       "${analyzer_header_text}")
string(REGEX MATCHALL "\\) noexcept" analyzer_noexcepts "${analyzer_header_text}")
list(LENGTH analyzer_declarations analyzer_declaration_count)
list(LENGTH analyzer_noexcepts analyzer_noexcept_count)
if(NOT analyzer_declaration_count EQUAL analyzer_noexcept_count)
    message(FATAL_ERROR
        "include/analyzer.h declares ${analyzer_declaration_count} functions but the C++ view "
        "marked ${analyzer_noexcept_count} noexcept; src/ffi/CMakeLists.txt needs updating")
endif()

# The C standard headers go in first, outside extern "C": libstdc++ and libc++
# wrap stdlib.h and others in templates, which cannot have C linkage. The header
# includes them again, and their guards make that a no-op.
file(CONFIGURE
     OUTPUT ${analyzer_ffi_generated}/ffi/c_api.hpp
     CONTENT "// Generated by src/ffi/CMakeLists.txt from include/analyzer.h. Do not edit.
#pragma once

#include <stdarg.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdlib.h>

extern \"C\" {
${analyzer_header_text}
}  // extern \"C\"
"
     @ONLY NEWLINE_STYLE UNIX)

analyzer_module(ffi
    DEPS analyzer::base analyzer::dsp analyzer::cal analyzer::plot analyzer::model
         analyzer::engine analyzer::audio)

# The public header is for C consumers. The generated C++ view is for this
# module and its tests, which call the entry points as Swift would.
target_include_directories(analyzer_ffi PUBLIC
    ${PROJECT_SOURCE_DIR}/include
    ${analyzer_ffi_generated})
